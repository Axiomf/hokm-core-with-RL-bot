cmake_minimum_required(VERSION 3.20)
project(hokm VERSION 0.1.0 LANGUAGES CXX)
option(HOKM_BUILD_PYTHON "Build Python bindings" OFF)
option(HOKM_BUILD_TESTS "Build correctness tests" ON)
option(HOKM_BUILD_TOOLS "Build demo and benchmark" ON)
add_library(hokm STATIC cpp/src/rules.cpp cpp/src/engine.cpp)
target_include_directories(hokm PUBLIC cpp/include)
target_compile_features(hokm PUBLIC cxx_std_17)
set_target_properties(hokm PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(MSVC)
 target_compile_options(hokm PRIVATE /W4)
else()
 target_compile_options(hokm PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(HOKM_BUILD_PYTHON)
 find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
 find_package(pybind11 CONFIG REQUIRED)
 pybind11_add_module(_core cpp/bindings/module.cpp)
 target_link_libraries(_core PRIVATE hokm)
 install(TARGETS _core LIBRARY DESTINATION hokm RUNTIME DESTINATION hokm)
endif()
if(HOKM_BUILD_TESTS)
 enable_testing()
 add_executable(hokm_tests tests/test_engine.cpp)
 target_link_libraries(hokm_tests PRIVATE hokm)
 add_test(NAME engine COMMAND hokm_tests)
endif()
if(HOKM_BUILD_TOOLS)
 add_executable(hokm_sim scripts/simulate.cpp)
 target_link_libraries(hokm_sim PRIVATE hokm)
endif()
